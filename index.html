💡 Mengapa Database Teks Anda Sangat Aman?

Untuk 40 karyawan, menyimpan teks absensi & izin selama 10 tahun penuh hanya memakan sekitar ~15 MB (kurang dari 3% kuota gratis 500 MB). Namun, fitur Kosongkan Transaksi di bawah ini sangat berguna untuk membersihkan log tahunan/bulanan setelah Anda mengunduh rekap Excel/CSV.

Sistem otomatis mengunduh file CSV berisi seluruh log absensi & pengajuan izin ke komputer Anda sebelum mengosongkan database. Data 40 karyawan & jadwal shift TIDAK akan terhapus.

Skema SQL PostgreSQL Supabase (Lengkap + Tabel Izin & Purge)

Salin dan jalankan perintah SQL ini di menu SQL Editor pada proyek Supabase Anda.

-- 1. Tabel Master Posisi & Pengaturan Jam Kerja
CREATE TABLE IF NOT EXISTS master_posisi (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  nama_posisi VARCHAR(120) NOT NULL,
  departemen VARCHAR(100) NOT NULL,
  target_jam_mingguan INT DEFAULT 40
);

-- 2. Tabel Master 40 Karyawan (Login Nama + PIN)
CREATE TABLE IF NOT EXISTS karyawan (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  nik VARCHAR(40) UNIQUE,
  nama VARCHAR(150) NOT NULL,
  pin_password VARCHAR(30) DEFAULT '123456',
  posisi VARCHAR(120) NOT NULL,
  level_posisi VARCHAR(50) DEFAULT 'Madya',
  departemen VARCHAR(100) NOT NULL,
  email VARCHAR(150),
  tanggal_bergabung DATE DEFAULT CURRENT_DATE,
  shift_default VARCHAR(20) DEFAULT 'SHIFT_1',
  avatar_style VARCHAR(40) DEFAULT 'female_bella',
  face_descriptor JSONB,
  jadwal_mingguan JSONB,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Tabel Log Absensi Wajah, Jam Kerja & Lokasi GPS
CREATE TABLE IF NOT EXISTS absensi (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  karyawan_id UUID REFERENCES karyawan(id) ON DELETE CASCADE,
  nama_karyawan VARCHAR(150),
  posisi_karyawan VARCHAR(120),
  tanggal DATE DEFAULT CURRENT_DATE,
  shift_kode VARCHAR(20) DEFAULT 'SHIFT_1',
  waktu_masuk TIMESTAMPTZ,
  waktu_pulang TIMESTAMPTZ,
  durasi_menit INT DEFAULT 0,
  durasi_teks VARCHAR(20) DEFAULT '08:00:00',
  menit_terlambat INT DEFAULT 0,
  jarak_meter INT DEFAULT 12,
  status_lokasi VARCHAR(50) DEFAULT 'Di Kantor (Valid)',
  status_kehadiran VARCHAR(40) DEFAULT 'Tepat Waktu',
  skor_wajah NUMERIC(5,2) DEFAULT 98.50,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Tabel Pengajuan Sakit / Izin / Cuti Karyawan
CREATE TABLE IF NOT EXISTS pengajuan_izin (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  karyawan_id UUID REFERENCES karyawan(id) ON DELETE CASCADE,
  nama_karyawan VARCHAR(150) NOT NULL,
  posisi_karyawan VARCHAR(120),
  jenis_pengajuan VARCHAR(50) NOT NULL, -- 'Sakit', 'Izin Keperluan', 'Cuti Tahunan', 'Dinas Luar'
  tanggal_mulai DATE NOT NULL,
  tanggal_selesai DATE NOT NULL,
  alasan TEXT NOT NULL,
  kontak_lampiran VARCHAR(150),
  status_persetujuan VARCHAR(40) DEFAULT 'Menunggu Persetujuan', -- 'Menunggu Persetujuan', 'Disetujui', 'Ditolak'
  approved_by VARCHAR(100),
  created_at TIMESTAMPTZ DEFAULT NOW()
);
